# --- FUNZIONI DI SUPPORTO ---
function Log-Gui {
    param([string]$Text, [string]$Color="Black")
    $rtbLog.SelectionStart = $rtbLog.Text.Length
    $rtbLog.SelectionColor = [System.Drawing.Color]::FromName($Color)
    $rtbLog.AppendText("[$((Get-Date).ToString('HH:mm:ss'))] $Text`r`n")
    $rtbLog.ScrollToCaret()
    [System.Windows.Forms.Application]::DoEvents()
}

function Wait-ProcessUI {
    param($Process)
    if ($null -eq $Process) { return $null }
    while (-not $Process.HasExited) {
        [System.Windows.Forms.Application]::DoEvents()
        Start-Sleep -Milliseconds 200
    }
    return $Process.ExitCode
}

# Download robusto usando curl nativo (bypass totale di Invoke-WebRequest e limiti .NET)
function Invoke-RobustDownload {
    param([string]$Url, [string]$OutFile)
    
    # Metodo 1: curl.exe nativo di Windows (Infallibile per TLS/Cipher)
    $curl = Get-Command "curl.exe" -ErrorAction SilentlyContinue
    if ($curl) {
        $args = @("-s", "-L", "-A", $UserAgent, "-o", $OutFile, $Url)
        $proc = Start-Process -FilePath "curl.exe" -ArgumentList $args -NoNewWindow -Wait -PassThru
        if ($proc.ExitCode -eq 0 -and (Test-Path $OutFile) -and ((Get-Item $OutFile).Length -gt 0)) { return $true }
    }

    # Metodo 2: Fallback su WebClient
    try {
        $wc = New-Object System.Net.WebClient
        $wc.Headers.Add("User-Agent", $UserAgent)
        $wc.DownloadFile($Url, $OutFile)
        if ((Test-Path $OutFile) -and ((Get-Item $OutFile).Length -gt 0)) { return $true }
    } catch {}

    # Metodo 3: Fallback finale su Invoke-WebRequest
    try {
        Invoke-WebRequest -Uri $Url -OutFile $OutFile -UseBasicParsing -UserAgent $UserAgent -ErrorAction Stop
        if ((Test-Path $OutFile) -and ((Get-Item $OutFile).Length -gt 0)) { return $true }
    } catch {}
    
    return $false
}

# Fetch testo robusto
function Get-RobustContent {
    param([string]$Url)
    $curl = Get-Command "curl.exe" -ErrorAction SilentlyContinue
    if ($curl) {
        try { 
            $output = & curl.exe -s -L -A $UserAgent $Url
            return $output -join "`n" 
        } catch {}
    }
    try { return (Invoke-WebRequest -Uri $Url -UseBasicParsing -UserAgent $UserAgent).Content } catch { return $null }
}

# Recupero dinamico link PDF24 dal changelog
function Get-Pdf24-Link {
    try {
        $ChangelogUrl = "https://creator.pdf24.org/changelog/en.html"
        $Content = Get-RobustContent -Url $ChangelogUrl
        if ($Content -match 'Version\s+(\d+\.\d+\.\d+)') {
            return "https://download.pdf24.org/pdf24-creator-$($matches[1])-x64.exe"
        }
        return $null
    } catch { return $null }
}

# Recupero dinamico link Kerio VPN Client dalle Release Notes
function Get-Kerio-Link {
    try {
        $Url = "https://www.gfi.com/products-and-solutions/network-security-solutions/kerio-control/release-notes"
        $Content = Get-RobustContent -Url $Url
        if ($Content -match 'Kerio Control\s+(\d+\.\d+\.\d+)\s+build\s+(\d+)') {
            $ver = $matches[1]
            $build = $matches[2]
            return "https://cdn.kerio.com/dwn/control/control-$ver-$build/kerio-control-vpnclient-$ver-$build-win64.msi"
        }
        return $null
    } catch { return $null }
}
